{{- /* Server-side math rendering via KaTeX (Hugo's transform.ToMath). Renders both inline (\( \)) and block ($$ … \[ \]) passthrough nodes at build time, so the browser never sees raw LaTeX. This is robust against HTML-special characters (<, >, &) that break client-side auto-render. */ -}} {{- $displayMode := ne .Type "inline" -}} {{- $opts := dict "displayMode" $displayMode "throwOnError" false "errorColor" "#cc0000" -}} {{- with try (transform.ToMath .Inner $opts) -}} {{- with .Err -}} {{- warnf "KaTeX render issue in %q: %s" $.Page.File.Path . -}} {{ $.Inner }} {{- else -}} {{- .Value -}} {{- end -}} {{- end -}}
